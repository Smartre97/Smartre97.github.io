c++增强三目运算符，使其可作为左值，返回表达式而非值，可赋值。

    int a=10;
    int b=20:
    int c=0;
    c =(a〈b)?a:b;
    cout <<c <endl;
    ((a<b)?a:b)=50;//三目运算符可以当左值，这里给a赋值50
    cout <<a<<endl;//输出50
